    \begin{minipage}{76mm}

        \begin{tabular}{lll}

        \textcolor{accent2}{$R_2$:} & $|C| > 0.15 N_{\text{total}} \;\Rightarrow$ & C (needs splitting) \\
        \textcolor{accent2}{$R_3$:} & $\text{keywords} \in \mathcal{B} \;\Rightarrow$ & D-meta (program marker) \\

        \textcolor{accent2}{$R_5$:} & $\bar{d}(C) > 1.5\bar{d}_{\text{median}} \;\Rightarrow$ & C (high variance) \\
        \textcolor{accent2}{$R_6$:} & \text{No rule triggered} \;\Rightarrow & \textbf{LLM Judge}
        \end{tabular}
    \end{minipage}
};
